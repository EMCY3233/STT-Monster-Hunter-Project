cmake_minimum_required(VERSION 3.16)
project(stt_container_inspector LANGUAGES CXX)
add_executable(stt-inspect main.cpp)
target_compile_features(stt-inspect PRIVATE cxx_std_17)
if(WIN32)
  target_link_libraries(stt-inspect PRIVATE bcrypt)
else()
  find_package(OpenSSL REQUIRED)
  target_link_libraries(stt-inspect PRIVATE OpenSSL::Crypto)
endif()
if(MSVC)
  target_compile_options(stt-inspect PRIVATE /W4 /WX)
else()
  target_compile_options(stt-inspect PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME synthetic COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_inspector.py $<TARGET_FILE:stt-inspect>)
