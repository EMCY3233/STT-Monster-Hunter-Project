cmake_minimum_required(VERSION 3.16)
project(stt_container_inspector LANGUAGES C CXX)
add_executable(stt-inspect main.cpp)
target_compile_features(stt-inspect PRIVATE cxx_std_17)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
include(FetchContent)
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(stt_zlib
  URL https://zlib.net/fossils/zlib-1.3.2.tar.gz
  URL_HASH SHA256=bb329a0a2cd0274d05519d61c667c062e06990d72e125ee2dfa8de64f0119d16
  TIMEOUT 60 TLS_VERIFY TRUE)
FetchContent_MakeAvailable(stt_zlib)
target_link_libraries(stt-inspect PRIVATE ZLIB::ZLIBSTATIC)

if(WIN32)
  target_link_libraries(stt-inspect PRIVATE bcrypt)
else()
  find_package(OpenSSL REQUIRED)
  target_link_libraries(stt-inspect PRIVATE OpenSSL::Crypto)
endif()
if(MSVC)
  target_compile_options(stt-inspect PRIVATE /W4 /WX)
else()
  target_compile_options(stt-inspect PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(STT_SANITIZERS "Enable ASan and UBSan on Linux" OFF)
if(STT_SANITIZERS)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "STT_SANITIZERS requires Linux")
  endif()
  target_compile_options(stt-inspect PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
  target_link_options(stt-inspect PRIVATE -fsanitize=address,undefined)
  target_compile_options(zlibstatic PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
endif()
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME synthetic COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_inspector.py $<TARGET_FILE:stt-inspect>)
