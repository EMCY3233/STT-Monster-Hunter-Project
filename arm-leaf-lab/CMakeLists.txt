cmake_minimum_required(VERSION 3.16)
project(stt_arm_leaf_lab LANGUAGES CXX)
set(STT_LEAF_INCLUDE "" CACHE PATH "Private directory containing translated.hpp")
if(STT_LEAF_INCLUDE)
  add_executable(stt_leaf_runner runner.cpp)
  target_include_directories(stt_leaf_runner PRIVATE "${STT_LEAF_INCLUDE}")
  target_compile_features(stt_leaf_runner PRIVATE cxx_std_17)
  set_target_properties(stt_leaf_runner PROPERTIES CXX_EXTENSIONS OFF)
  if(MSVC)
    target_compile_options(stt_leaf_runner PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(stt_leaf_runner PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endif()
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(BUILD_TESTING)
  add_test(NAME synthetic_translation
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_translation.py"
      --cmake "${CMAKE_COMMAND}" --config "$<CONFIG>")
  set_tests_properties(synthetic_translation PROPERTIES TIMEOUT 180)
endif()
