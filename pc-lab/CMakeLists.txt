cmake_minimum_required(VERSION 3.16)
project(stt_pc_lab LANGUAGES CXX)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
add_library(stt_pc_core src/simulation.cpp)
target_include_directories(stt_pc_core PUBLIC include)
target_compile_features(stt_pc_core PUBLIC cxx_std_17)
set_target_properties(stt_pc_core PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(stt_pc_core PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(stt_pc_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(stt_pc_demo src/main.cpp)
target_link_libraries(stt_pc_demo PRIVATE stt_pc_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(stt_pc_tests tests/simulation_tests.cpp)
  target_link_libraries(stt_pc_tests PRIVATE stt_pc_core)
  add_test(NAME simulation_contract COMMAND stt_pc_tests)
  add_test(NAME synthetic_demo COMMAND stt_pc_demo)
  add_executable(stt_arena_tests tests/arena_tests.cpp)
  target_link_libraries(stt_arena_tests PRIVATE stt_pc_core)
  add_test(NAME arena_contract COMMAND stt_arena_tests)
endif()

option(STT_NATIVE_ARENA "Build original OpenGL 3.3 native arena" OFF)
if(STT_NATIVE_ARENA)
  include(FetchContent)
  set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(glfw
    URL https://codeload.github.com/glfw/glfw/zip/refs/tags/3.4
    URL_HASH SHA256=a133ddc3d3c66143eba9035621db8e0bcf34dba1ee9514a9e23e96afd39fd57a)
  FetchContent_MakeAvailable(glfw)
  find_package(OpenGL REQUIRED)
  add_executable(stt_native_arena src/native_arena.cpp)
  target_link_libraries(stt_native_arena PRIVATE stt_pc_core glfw OpenGL::GL)
  if(MSVC)
    target_compile_options(stt_native_arena PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(stt_native_arena PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  if(WIN32)
    target_compile_definitions(stt_native_arena PRIVATE
      WINVER=0x0A00 _WIN32_WINNT=0x0A00 NOMINMAX)
  endif()
  if(BUILD_TESTING)
    add_test(NAME native_render_smoke COMMAND stt_native_arena --smoke)
    set_tests_properties(native_render_smoke PROPERTIES TIMEOUT 45)
  endif()
endif()
