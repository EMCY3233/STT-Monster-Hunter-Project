cmake_minimum_required(VERSION 3.18)
project(stt_combat_lab C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(stt_combat STATIC src/stt_combat.c)
target_include_directories(stt_combat PUBLIC include)
if(MSVC)
 target_compile_options(stt_combat PRIVATE /W4)
else()
 target_compile_options(stt_combat PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_executable(stt_combat_demo src/demo.c)
target_link_libraries(stt_combat_demo PRIVATE stt_combat)
add_executable(stt_combat_test tests/test_combat.c)
target_link_libraries(stt_combat_test PRIVATE stt_combat)
# Assertions remain enabled in release test builds.
if(MSVC)
 target_compile_options(stt_combat_test PRIVATE /UNDEBUG)
else()
 target_compile_options(stt_combat_test PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME combat COMMAND stt_combat_test)
add_test(NAME synthetic_demo COMMAND stt_combat_demo)
