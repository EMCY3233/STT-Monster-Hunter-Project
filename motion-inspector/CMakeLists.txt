cmake_minimum_required(VERSION 3.20)
project(stt_motion LANGUAGES CXX)
add_executable(stt-motion main.cpp)
target_compile_features(stt-motion PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(stt-motion PRIVATE /W4 /WX)
else()
  target_compile_options(stt-motion PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(STT_MOTION_SANITIZERS "Enable ASan/UBSan" OFF)
if(STT_MOTION_SANITIZERS AND NOT MSVC)
  target_compile_options(stt-motion PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  target_link_options(stt-motion PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME synthetic COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_motion.py" "$<TARGET_FILE:stt-motion>")
