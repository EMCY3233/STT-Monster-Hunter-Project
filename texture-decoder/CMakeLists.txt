cmake_minimum_required(VERSION 3.20)
project(stt_tex LANGUAGES CXX)
add_executable(stt-tex main.cpp)
target_compile_features(stt-tex PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(stt-tex PRIVATE /W4 /WX)
else()
  target_compile_options(stt-tex PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME synthetic COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_tex.py" "$<TARGET_FILE:stt-tex>")
