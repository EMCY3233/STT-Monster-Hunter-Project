cmake_minimum_required(VERSION 3.20)
project(stt_skin LANGUAGES CXX)
add_executable(stt-skin main.cpp)
target_compile_features(stt-skin PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(stt-skin PRIVATE /W4 /WX)
else()
  target_compile_options(stt-skin PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(STT_SKIN_SANITIZERS "Enable ASan/UBSan" OFF)
if(STT_SKIN_SANITIZERS AND NOT MSVC)
  target_compile_options(stt-skin PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  target_link_options(stt-skin PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME synthetic COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_skin.py" "$<TARGET_FILE:stt-skin>")
add_test(NAME motion_simulation COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/simulate_motion.py")
