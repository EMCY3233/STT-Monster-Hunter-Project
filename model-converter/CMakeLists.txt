cmake_minimum_required(VERSION 3.20)
project(stt_model LANGUAGES CXX)
add_executable(stt-model main.cpp)
target_compile_features(stt-model PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(stt-model PRIVATE /W4 /WX)
else()
  target_compile_options(stt-model PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(STT_MODEL_SANITIZERS "Enable ASan/UBSan" OFF)
if(STT_MODEL_SANITIZERS AND NOT MSVC)
  target_compile_options(stt-model PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  target_link_options(stt-model PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME synthetic COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_model.py" "$<TARGET_FILE:stt-model>")
add_test(NAME inspector COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_inspector.py")
